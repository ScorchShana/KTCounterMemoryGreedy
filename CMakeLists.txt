option(ENABLE_ARENA_ASAN_TESTS "Build Arena Address/UndefinedBehaviorSanitizer tests" OFF)
foreach(ARENA_SANITIZER IN ITEMS asan tsan)
    if((ARENA_SANITIZER STREQUAL "asan" AND ENABLE_ARENA_ASAN_TESTS)
        OR (ARENA_SANITIZER STREQUAL "tsan" AND ENABLE_TSAN_TESTS))
        if(ARENA_SANITIZER STREQUAL "asan")
            set(ARENA_SANITIZER_FLAGS -fsanitize=address,undefined)
        else()
            set(ARENA_SANITIZER_FLAGS -fsanitize=thread)
        endif()
        foreach(ARENA_TEST IN ITEMS concurrent_memory_pool first_memory_scan first_pool_lifecycle tree_allocation_failure first_spill_writer classifier_first_batch low_frequency_io msd_prefix_codec)
            set(ARENA_TARGET arena_${ARENA_TEST}_${ARENA_SANITIZER}_test)
            add_executable(${ARENA_TARGET} tests/test_${ARENA_TEST}.cpp)
            target_compile_definitions(${ARENA_TARGET} PRIVATE TEST_MODE)
            target_compile_options(${ARENA_TARGET} PRIVATE -O1 -g ${ARENA_SANITIZER_FLAGS} -fno-omit-frame-pointer -fno-lto)
            set_property(TARGET ${ARENA_TARGET} APPEND_STRING PROPERTY LINK_FLAGS " ${ARENA_SANITIZER_FLAGS} -fno-lto")
            target_link_libraries(${ARENA_TARGET} pthread rt atomic)
            if(ARENA_TEST STREQUAL "classifier_first_batch")
                target_link_libraries(${ARENA_TARGET} zlib)
            endif()
            if(NUMA_LIBRARY)
                target_link_libraries(${ARENA_TARGET} ${NUMA_LIBRARY})
            endif()
            set(ARENA_TEST_ARGS)
            if(ARENA_TEST STREQUAL "low_frequency_io" AND ARENA_SANITIZER STREQUAL "tsan")
                set(ARENA_TEST_ARGS --no-death)
            endif()
            if(ARENA_SANITIZER STREQUAL "tsan" AND SETARCH_EXECUTABLE)
                add_test(NAME ${ARENA_TARGET} COMMAND ${SETARCH_EXECUTABLE} ${CMAKE_SYSTEM_PROCESSOR} -R $<TARGET_FILE:${ARENA_TARGET}> ${ARENA_TEST_ARGS})
            else()
                add_test(NAME ${ARENA_TARGET} COMMAND ${ARENA_TARGET} ${ARENA_TEST_ARGS})
            endif()
            set_tests_properties(${ARENA_TARGET} PROPERTIES LABELS ${ARENA_SANITIZER} TIMEOUT 120)
            if(ARENA_TEST STREQUAL "classifier_first_batch")
                set_tests_properties(${ARENA_TARGET} PROPERTIES TIMEOUT 240)
            elseif(ARENA_TEST STREQUAL "msd_prefix_codec")
                set_tests_properties(${ARENA_TARGET} PROPERTIES TIMEOUT 600)
            endif()
        endforeach()
    endif()
endforeach()
